   173	func parseBackendStates(output string) map[string]string {
   174		states := map[string]string{}
   175		currentHost := "unknown"
   176		var block strings.Builder
   177		flush := func() {
   178			if block.Len() == 0 {
   179				return
   180			}
   181			reader := csv.NewReader(strings.NewReader(block.String()))
   182			reader.FieldsPerRecord = -1
   183			var header []string
   184			for {
   185				row, err := reader.Read()
   186				if err != nil {
   187					break
   188				}
   189				if len(row) == 0 {
   190					continue
   191				}
   192				if strings.TrimSpace(row[0]) == "# pxname" {
   193					header = row
   194					continue
   195				}
   196				if len(header) == 0 || len(row) < len(header) {
   197					continue
   198				}
   199				indexes := map[string]int{}
   200				for i, h := range header {
   201					indexes[strings.TrimSpace(h)] = i
   202				}
   203				px, pxOK := indexes["# pxname"]
   204				if !pxOK {
   205					px, pxOK = indexes["pxname"]
   206				}
   207				sv, svOK := indexes["svname"]
   208				status, statusOK := indexes["status"]
   209				if !pxOK || !svOK || !statusOK || px >= len(row) || sv >= len(row) || status >= len(row) {
   210					continue
   211				}
   212				name := strings.TrimSpace(row[sv])
   213				if name != "BACKEND" && name != "FRONTEND" {
   214					states[currentHost+"/"+strings.TrimSpace(row[px])+"/"+name] = strings.TrimSpace(row[status])
   215				}
   216			}
   217			block.Reset()
   218		}
   219		for _, line := range strings.Split(output, "\n") {
   220			if strings.Contains(line, " | ") && strings.HasSuffix(strings.TrimSpace(line), ">>") {
   221				flush()
   222				currentHost = strings.SplitN(line, " | ", 2)[0]
   223				continue
   224			}
   225			if strings.Contains(line, " | FAILED | ") || strings.Contains(line, " | UNREACHABLE!") {
   226				flush()
   227				continue
   228			}
   229			block.WriteString(line)
   230			block.WriteByte('\n')
   231		}
   232		flush()
   233		return states
   234	}

     1	# Test Plan
     2	
     3	## DEFERRED TESTS
     4	
     5	- **Live VRRP failover**: In an isolated staging environment, set `GATEWAY_API_BASE_URL` (default `http://localhost:8081`), `CONTROL_PLANE_TOKEN` (when enabled), `FAILOVER_PROBE_URL` to a routed HTTP service hostname, and `FAILOVER_THRESHOLD_MS` (default 3000). Run `scripts/failover-test.sh`. Confirm that the status API's `vip_holder` before and after differs, the VIP response has the matching `X-Gateway-ID`, service returned HTTP 2xx/3xx within the threshold, keepalived is restored by the EXIT trap, and the `failover` event has a measured duration.
     6	- **Control-plane failover endpoint**: POST `{"confirm":true}` to `/api/gateways/failover` in staging. Verify the active gateway is stopped and restarted through Ansible, the VIP responds from the peer, concurrent requests receive 409, and the timeout path returns 504 while still restoring keepalived.
     7	- **HAProxy identity header deployment**: Apply the HAProxy role to both staging gateways and verify public HTTP and HTTPS responses contain `X-Gateway-ID: gw1` or `gw2`; verify the loopback health frontend on `127.0.0.1:9000` does not expose the public header behavior.
     8	- **Shell failover test execution**: Run `scripts/run-all-tests.sh --test keepalived-weight` only as a static priority/weight check. It does not exercise real failover; the live check remains `scripts/failover-test.sh` and is intentionally excluded from the default test suite.
     9	
    10	## DEFERRED TESTS — Remote HAProxy rollback
    11	
    12	- **Valid apply**: In isolated staging, POST `/api/gateways/sync`; verify both candidate files pass remote `haproxy -c`, backups exist on each gateway for the returned generation, standby reloads first, both services remain active, and the generation commits.
    13	- **Broken config rejected**: Inject an invalid candidate in a staging-only test. Verify remote validation fails on either gateway before any live file is replaced or service reloaded.
    14	- **Unhealthy config auto-rollback**: Deploy a candidate that causes a backend regression; verify health detection identifies the newly DOWN backend during `HEALTH_REGRESSION_WINDOW_SECONDS`, a remote rollback is staged/validated/reloaded, and an `auto` rollback event records reason and duration.
    15	- **Manual rollback checksum**: POST `/api/gateways/rollback` with an optional `target_generation`; compare `sha256sum /etc/haproxy/haproxy.cfg` on gw1 and gw2 against the selected committed snapshot after host-specific `X-Gateway-ID` insertion, and verify real duration/event fields.
    16	- **Operation mutex**: Hold a sync or rollback operation open, then POST failover/sync/rollback concurrently; verify the competing gateway operation receives HTTP 409.
